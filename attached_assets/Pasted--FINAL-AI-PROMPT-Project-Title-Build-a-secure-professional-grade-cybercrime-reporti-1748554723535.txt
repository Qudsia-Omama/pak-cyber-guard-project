```jsx
function ReportForm({ isAnonymous, setIsAnonymous }) {
  const [step, setStep] = useState(1)
  const [errors, setErrors] = useState({})
  const [isSubmitting, setIsSubmitting] = useState(false)
  const [reportId, setReportId] = useState(null)
  const [formData, setFormData] = useState({
    crimeType: '',
    description: '',
    date: '',
    time: '',
    evidence: [],
    firstName: '',
    lastName: '',
    email: '',
    phone: '',
    consent: false
  })

  const crimeTypes = [
    { id: 'phishing', name: 'Phishing', icon: '🎣', description: 'Fake emails/websites stealing info' },
    { id: 'financial_fraud', name: 'Financial Fraud', icon: '💳', description: 'Online scams, fake investments' },
    { id: 'harassment', name: 'Harassment', icon: '📱', description: 'Cyberbullying, threats, stalking' },
    { id: 'hacking', name: 'Hacking', icon: '🔓', description: 'Unauthorized account access' },
    { id: 'other', name: 'Other', icon: '📝', description: 'Different type of cyber crime' }
  ]

  const validateEmail = (email) => {
    const re = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    return re.test(email)
  }

  const validatePhone = (phone) => {
    const re = /^[+]?[(]?[\d\s\-()]{10,}$/
    return re.test(phone.replace(/\s/g, ''))
  }

  const validateStep = (stepNumber) => {
    const newErrors = {}

    switch (stepNumber) {
      case 1:
        if (!formData.crimeType) {
          newErrors.crimeType = 'Please select a crime type'
        }
        break
      case 2:
        if (!formData.date) {
          newErrors.date = 'Please select the incident date'
        } else {
          const selectedDate = new Date(formData.date)
          const today = new Date()
          if (selectedDate > today) {
            newErrors.date = 'Incident date cannot be in the future'
          }
        }
        if (!formData.description.trim()) {
          newErrors.description = 'Please describe the incident'
        } else if (formData.description.trim().length < 20) {
          newErrors.description = 'Please provide more details (at least 20 characters)'
        }
        break
      case 3:
        if (!isAnonymous) {
          if (formData.email && !validateEmail(formData.email)) {
            newErrors.email = 'Please enter a valid email address'
          }
          if (formData.phone && !validatePhone(formData.phone)) {
            newErrors.phone = 'Please enter a valid phone number'
          }
        }
        break
      case 5:
        if (!formData.consent) {
          newErrors.consent = 'Please confirm the accuracy of your information'
        }
        break
    }

    setErrors(newErrors)
    return Object.keys(newErrors).length === 0
  }

  const handleFileUpload = (e) => {
    const files = Array.from(e.target.files)
    const maxSize = 5 * 1024 * 1024 // 5MB
    const allowedTypes = ['image/', 'video/', 'application/pdf', 'application/msword', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document']

    const validFiles = files.filter(file => {
      if (file.size > maxSize) {
        showAlert(File ${file.name} is too large. Maximum size is 5MB., 'error')
        return false
      }
      if (!allowedTypes.some(type => file.type.startsWith(type) || file.type === type)) {
        showAlert(File ${file.name} type not supported., 'error')
        return false
      }
      return true
    })

    if (validFiles.length > 0) {
      setFormData(prev => ({ ...prev, evidence: [...prev.evidence, ...validFiles] }))
    }
  }

  const removeFile = (index) => {
    setFormData(prev => ({
      ...prev,
      evidence: prev.evidence.filter((_, i) => i !== index)
    }))
  }

  const nextStep = () => {
    if (validateStep(step)) {
      setStep(prev => Math.min(prev + 1, 6))
    } else {
      showAlert('Please fix the errors before continuing', 'error')
    }
  }

  const prevStep = () => setStep(prev => Math.max(prev - 1, 1))

  const submitReport = async () => {
    if (!validateStep(5)) {
      showAlert('Please fix the errors before submitting', 'error')
      return
    }

    setIsSubmitting(true)

    try {
      // Simulate API call
      await new Promise(resolve => setTimeout(resolve, 2000))

      // Generate a mock report ID
      const mockReportId = CR-${Date.now().toString().slice(-6)}
      setReportId(mockReportId)

      setStep(6)
      showAlert('Report submitted successfully!', 'success')
    } catch (error) {
      showAlert('Failed to submit report. Please try again.', 'error')
    } finally {
      setIsSubmitting(false)
    }
  }

  const resetForm = () => {
    setFormData({
      crimeType: '',
      description: '',
      date: '',
      time: '',
      evidence: [],
      firstName: '',
      lastName: '',
      email: '',
      phone: '',
      consent: false
    })
    setErrors({})
    setStep(1)
    setReportId(null)
    setIsAnonymous(false)
  }

  const showAlert = (message, type) => {
    // Create alert element
    const alert = document.createElement('div')
    alert.className = alert ${type === 'error' ? 'error' : 'success'}
    alert.innerHTML = `
      <span class="alert-icon">${type === 'error' ? '⚠️' : '✅'}</span>
      <span>${message}</span>
    `

    // Add to document
    document.body.appendChild(alert)

    // Show and auto-hide
    setTimeout(() => alert.classList.add('show'), 100)
    setTimeout(() => {
      alert.classList.remove('show')
      setTimeout(() => alert.remove(), 300)
    }, 4000)
  }

  const getSelectedCrimeType = () => {
    return crimeTypes.find(type => type.id === formData.crimeType)
  }

  return (
    <div className="report-form">
      <div className="form-header">
        <h2>Report Cybercrime</h2>
        <div className="progress-bar">
          <div className="progress" style={{ width: ${Math.min((step / 6) * 100, 100)}% }}></div>
        </div>
        <div className="step-indicator">Step {Math.min(step, 5)} of 5</div>
      </div>

      <div className="anonymous-toggle">
        <label className="toggle-switch">
          <input 
            type="checkbox" 
            checked={isAnonymous}
            onChange={(e) => setIsAnonymous(e.target.checked)}
          />
          <span className="slider"></span>
          Report Anonymously
        </label>
      </div>

      {step === 1 && (
        <div className="step-content">
          <h3>What type of cyber crime are you reporting?</h3>
          <div className="crime-types-grid">
            {crimeTypes.map(type => (
              <div 
                key={type.id}
                className={crime-card ${formData.crimeType === type.id ? 'selected' : ''}}
                onClick={() => setFormData(prev => ({ ...prev, crimeType: type.id }))}
              >
                <div className="crime-icon">{type.icon}</div>
                <h4>{type.name}</h4>
                <p className="crime-description">{type.description}</p>
              </div>
            ))}
          </div>
          {errors.crimeType && <div className="error-message">{errors.crimeType}</div>}
        </div>
      )}

      {step === 2 && (
        <div className="step-content">
          <h3>Provide details about the incident</h3>
          <div className="form-group">
            <label>When did this happen? <span className="required">*</span></label>
            <input 
              type="date" 
              value={formData.date}
              max={new Date().toISOString().split('T')[0]}
              onChange={(e) => setFormData(prev => ({ ...prev, date: e.target.value }))}
              className={errors.date ? 'error' : ''}
            />
            {errors.date && <div className="error-message">{errors.date}</div>}
          </div>
          <div className="form-group">
            <label>Time of Incident (optional)</label>
            <input 
              type="time" 
              value={formData.time}
              onChange={(e) => setFormData(prev => ({ ...prev, time: e.target.value }))}
            />
          </div>
          <div className="form-group">
            <label>Description of what happened <span className="required">*</span></label>
            <textarea 
              rows="6"
              placeholder="Provide as many details as possible about the incident..."
              value={formData.description}
              onChange={(e) => setFormData(prev => ({ ...prev, description: e.target.value }))}
              className={errors.description ? 'error' : ''}
            />
            <div className="char-count">{formData.description.length} characters</div>
            {errors.description && <div className="error-message">{errors.description}</div>}
          </div>
        </div>
      )}

      {step === 3 && (
        <div className="step-content">
          <h3>Your contact information {isAnonymous ? '(Skipped - Anonymous Report)' : '(Optional)'}</h3>
          {isAnonymous ? (
            <div className="anonymous-notice">
              <div className="notice-icon">🔒</div>
              <p>You've chosen to report anonymously. Your personal information will not be collected.</p>
            </div>
          ) : (
            <>
              <p className="step-description">Providing contact information allows us to follow up if needed. This information will be kept confidential.</p>
              <div className="form-row">
                <div className="form-group">
                  <label>First Name</label>
                  <input 
                    type="text" 
                    value={formData.firstName}
                    onChange={(e) => setFormData(prev => ({ ...prev, firstName: e.target.value }))}
                    placeholder="Enter your first name"
                  />
                </div>
                <div className="form-group">
                  <label>Last Name</label>
                  <input 
                    type="text" 
                    value={formData.lastName}
                    onChange={(e) => setFormData(prev => ({ ...prev, lastName: e.target.value }))}
                    placeholder="Enter your last name"
                  />
                </div>
              </div>
              <div className="form-group">
                <label>Email Address</label>
                <input 
                  type="email" 
                  value={formData.email}
                  onChange={(e) => setFormData(prev => ({ ...prev, email: e.target.value }))}
                  placeholder="your.email@example.com"
                  className={errors.email ? 'error' : ''}
                />
                {errors.email && <div className="error-message">{errors.email}</div>}
              </div>
              <div className="form-group">
                <label>Phone Number</label>
                <input 
                  type="tel" 
                  value={formData.phone}
                  onChange={(e) => setFormData(prev => ({ ...prev, phone: e.target.value }))}
                  placeholder="+1 (555) 123-4567"
                  className={errors.phone ? 'error' : ''}
                />
                {errors.phone && <div className="error-message">{errors.phone}</div>}
              </div>
            </>
          )}
        </div>
      )}

      {step === 4 && (
        <div className="step-content">
          <h3>Upload Evidence (Optional)</h3>
          <p className="step-description">Upload any screenshots, documents, or other evidence related to the incident.</p>
          <div className="upload-zone">
            <input 
              type="file" 
              multiple 
              onChange={handleFileUpload}
              accept="image/,video/,.pdf,.doc,.docx"
              id="evidence-upload"
            />
            <label htmlFor="evidence-upload" className="upload-label">
              <div className="upload-icon">📎</div>
              <p>Drag and drop files or click to browse</p>
              <p className="upload-hint">Images, videos, documents accepted (Max 5MB each)</p>
            </label>
          </div>
          {formData.evidence.length > 0 && (
            <div className="evidence-list">
              <h4>Uploaded Files ({formData.evidence.length}):</h4>
              {formData.evidence.map((file, index) => (
                <div key={index} className="evidence-item">
                  <div className="file-info">
                    <span className="file-name">{file.name}</span>
                    <span className="file-size">({(file.size / 1024 / 1024).toFixed(2)} MB)</span>
                  </div>
                  <button 
                    type="button"
                    className="remove-btn"
                    onClick={() => removeFile(index)}
                  >
                    Remove
                  </button>
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {step === 5 && (
        <div className="step-content">
          <h3>Review your report</h3>
          <p className="step-description">Please review all information before submitting your report.</p>

          <div className="review-section">
            <div className="review-card">
              <h4>Incident Details</h4>
              <div className="review-item">
                <span className="label">Crime Type:</span>
                <span className="value">{getSelectedCrimeType()?.name || '-'}</span>
              </div>
              <div className="review-item">
                <span className="label">Date:</span>
                <span className="value">{formData.date || '-'}</span>
              </div>
              {formData.time && (
                <div className="review-item">
                  <span className="label">Time:</span>
                  <span className="value">{formData.time}</span>
                </div>
              )}
              <div className="review-item">
                <span className="label">Description:</span>
                <span className="value description">{formData.description || '-'}</span>
              </div>
              <div className="review-item">
                <span className="label">Evidence Files:</span>
                <span className="value">{formData.evidence.length > 0 ? ${formData.evidence.length} file(s) : 'None'}</span>
              </div>
            </div>

            <div className="review-card">
              <h4>Contact Information</h4>
              {isAnonymous ? (
                <div className="anonymous-notice">
                  <span className="value">Anonymous Report - No contact information provided</span>
                </div>
              ) : (
                <>
                  {(formData.firstName || formData.lastName) && (
                    <div className="review-item">
                      <span className="label">Name:</span>
                      <span className="value">{${formData.firstName} ${formData.lastName}.trim() || '-'}</span>
                    </div>
                  )}
                  {formData.email && (
                    <div className="review-item">
                      <span className="label">Email:</span>
                      <span className="value">{formData.email}</span>
                    </div>
                  )}
                  {formData.phone && (
                    <div className="review-item">
                      <span className="label">Phone:</span>
                      <span className="value">{formData.phone}</span>
                    </div>
                  )}
                  {!formData.firstName && !formData.lastName && !formData.email && !formData.phone && (
                    <div className="review-item">
                      <span className="value">No contact information provided</span>
                    </div>
                  )}
                </>
              )}
            </div>
          </div>

          <div className="consent-section">
            <label className="consent-checkbox">
              <input 
                type="checkbox" 
                checked={formData.consent}
                onChange={(e) => setFormData(prev => ({ ...prev, consent: e.target.checked }))}
              />
              <span className="checkmark"></span>
              I confirm that the information provided is accurate to the best of my knowledge.
            </label>
            {errors.consent && <div className="error-message">{errors.consent}</div>}
          </div>
        </div>
      )}

      {step === 6 && (
        <div className="step-content success-step">
          <div className="success-icon">✅</div>
          <h3>Report Submitted Successfully!</h3>
          <p className="success-message">Thank you for helping make the internet safer. Your report has been received and will be processed by our team.</p>

          {reportId && (
            <div className="report-id">
              <strong>Report ID: {reportId}</strong>
              <p>Save this ID for your records</p>
            </div>
          )}

          <div className="next-steps">
            <h4>What happens next?</h4>
            <ul>
              <li>You'll receive a confirmation email with your case reference number</li>
              <li>Our system will analyze your report and categorize it</li>
              <li>If additional information is needed, we may contact you (unless anonymous)</li>
              <li>Depending on the case, we may forward it to the appropriate authorities</li>
            </ul>
          </div>

          <div className="success-actions">
            <button className="btn btn-primary" onClick={resetForm}>
              Submit Another Report
            </button>
          </div>
        </div>
      )}

      {step < 6 && (
        <div className="form-actions">
          {step > 1 && (
            <button className="btn btn-secondary" onClick={prevStep} disabled={isSubmitting}>
              Previous
            </button>
          )}
          {step < 5 ? (
            <button className="btn btn-primary" onClick={nextStep} disabled={isSubmitting}>
              Next
            </button>
          ) : (
            <button 
              className="btn btn-primary" 
              onClick={submitReport}
              disabled={isSubmitting}
            >
              {isSubmitting ? 'Submitting...' : 'Submit Report'}
            </button>
          )}
        </div>
      )}
    </div>
  )
}
